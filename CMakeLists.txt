cmake_minimum_required(VERSION 3.16)
project(corearena VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(COREARENA_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(corearena src/corearena.c)
target_include_directories(corearena PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(corearena PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
  if(COREARENA_SANITIZE)
    target_compile_options(corearena PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(corearena PUBLIC -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_executable(corearena_tests tests/test_corearena.c)
target_link_libraries(corearena_tests PRIVATE corearena)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(corearena_tests PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_test(NAME corearena_tests COMMAND corearena_tests)
add_executable(frame_example examples/frame.c)
target_link_libraries(frame_example PRIVATE corearena)
add_executable(corearena_bench benchmarks/bench.c)
target_link_libraries(corearena_bench PRIVATE corearena)
install(TARGETS corearena ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(FILES include/corearena.h DESTINATION include)
